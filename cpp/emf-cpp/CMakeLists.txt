if(EMF_BUILD_TESTS)
    enable_testing()
endif()

# Helper: only add subdirectory if all .cpp files referenced in its CMakeLists.txt actually exist.
function(emf_add_optional_subdir subdir)
    if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${subdir}/CMakeLists.txt)
        return()
    endif()
    file(READ ${CMAKE_CURRENT_SOURCE_DIR}/${subdir}/CMakeLists.txt _emf_cmake)
    string(REGEX MATCHALL "src/[A-Za-z0-9_]+\\.cpp" _emf_refs "${_emf_cmake}")
    set(_emf_missing 0)
    foreach(_ref IN LISTS _emf_refs)
        if(NOT EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${subdir}/${_ref})
            set(_emf_missing 1)
            break()
        endif()
    endforeach()
    if(_emf_missing)
        message(STATUS "Skipping subproject ${subdir}: missing source file(s)")
        return()
    endif()
    add_subdirectory(${subdir})
endfunction()
